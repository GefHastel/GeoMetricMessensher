# ===========================================================================
#  GeoMetric — библиотеки для запуска ВСЕГО из исходников.
#  Сюда входит и сервер (server.py), и приложение для компьютера (папка desktop).
#
#  Как поставить:      python -m pip install -r requirements.txt
#  Нужен Python 3.10 или новее — в коде используются современные подсказки типов.
#  Для сборки архива GeoMetric-Windows-x64.zip версии библиотек закрепляются
#  отдельно и жёстко — они перечислены в файле make_win_package.py.
# ===========================================================================

# ------------------------- СЕРВЕР (server.py) -------------------------
Flask>=3.0                      # сам веб-сервер: принимает запросы, отдаёт файлы и API (вход, поиск, отправка)
Flask-SocketIO>=5.3             # «живой» канал: мгновенная доставка сообщений, «печатает…», сигналы звонков
simple-websocket>=1.0           # поддержка WebSocket: без неё Socket.IO работает медленнее — «опросами»

# ------------- ПРИЛОЖЕНИЕ ДЛЯ КОМПЬЮТЕРА (папка desktop/) -------------
PySide6-Essentials>=6.7         # настоящие окна, кнопки и списки (Qt). Без лишних модулей — сборка легче
python-socketio[client]>=5.11   # клиент Socket.IO: держит живую связь приложения с сервером
websocket-client>=1.7           # транспорт WebSocket для этого клиента (быстрый канал вместо опросов)
cryptography>=42.0              # сквозное шифрование: AES-256-GCM и обмен ключами ECDH (кривая P-256)

# ---- Эти библиотеки ставятся сами вместе с верхними, указывать не нужно ----
# Werkzeug, Jinja2, click, itsdangerous, blinker — приходят вместе с Flask
# python-engineio, bidict — приходят вместе с python-socketio
# shiboken6 — приходит вместе с PySide6
# cffi, pycparser — приходят вместе с cryptography
